git clone <Clone_url> # Clone a remote repository.
git status # Show status of working directory.
git add <file> # Stage a file for commit.
git add . # Stage all changes.
git commit -m "message" # Stage commit changes with message.
git branch # list all branches.
git branch <branch_name> # Create a new Branch.
git checkout -b <branch_name> # Switch to a Branch.
git merge <branch_name> # Merge branch file to the branch.
git branch -d <branch_name> # Delete a Branch.
git push # Uploads local commits to repository.
git log # Displays the commit history.
